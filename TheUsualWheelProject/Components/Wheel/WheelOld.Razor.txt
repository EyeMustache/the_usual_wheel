@using SkiaSharp
@using SkiaSharp.Views.Blazor
@inject WheelViewModel ViewModel

<div class="wheel-wrapper" style="width: 300px; height: 300px; margin: 0 auto; position: relative; background-color: #f0f0f0; border-radius: 50%;">
    <SKCanvasView @ref="_canvasView" OnPaintSurface="OnPaintSurface" style="width: 100%; height: 100%;" IgnorePixelScaling="true" />
    
    @* Center Pin/Indicator *@
    <div style="position: absolute; top: -10px; left: 50%; transform: translateX(-50%); width: 0; height: 0; border-left: 15px solid transparent; border-right: 15px solid transparent; border-top: 30px solid red; z-index: 10;"></div>
</div>

<div class="controls mt-3 text-center">
    <button class="btn btn-primary" @onclick="SpinAsync" disabled="@_isSpinning">
        @(_isSpinning ? "Spinning..." : "SPIN THE WHEEL")
    </button>
</div>

@code {
    private SKCanvasView? _canvasView;
    private bool _isSpinning = false;
    private float _rotationAngle = 0f;
    private float _spinVelocity = 0f;

    // Safely get active movies, fallback to empty list if WheelMovies is null
    private List<WheelMovie> GetActiveMovies()
    {
        if (ViewModel.WheelMovies == null) return new();
        return ViewModel.WheelMovies.Where(m => !m.IsEliminated).ToList();
    }

    private void OnPaintSurface(SKPaintSurfaceEventArgs e)
    {
        var canvas = e.Surface.Canvas;
        canvas.Clear(SKColors.Transparent);

        var activeMovies = GetActiveMovies();
        int count = activeMovies.Count;
        
        // --- Empty State ---
        if (count == 0)
        {
            var emptyCenter = new SKPoint(e.Info.Rect.MidX, e.Info.Rect.MidY);
            float emptyRadius = Math.Min(e.Info.Width, e.Info.Height) / 2f * 0.9f;
            using var emptyPaint = new SKPaint { Color = SKColors.LightGray, Style = SKPaintStyle.Fill, IsAntialias = true };
            canvas.DrawCircle(emptyCenter, emptyRadius, emptyPaint);
            return;
        }

        // --- Populated State ---
        float sweepAngle = 360f / count;
        float radius = Math.Min(e.Info.Width, e.Info.Height) / 2f * 0.9f;
        var center = new SKPoint(e.Info.Rect.MidX, e.Info.Rect.MidY);
        var rect = new SKRect(center.X - radius, center.Y - radius, center.X + radius, center.Y + radius);

        canvas.Save();
        canvas.RotateDegrees(_rotationAngle, center.X, center.Y);

        for (int i = 0; i < count; i++)
        {
            // Draw Segment
            using var path = new SKPath();
            path.MoveTo(center);
            path.ArcTo(rect, i * sweepAngle, sweepAngle, false);
            path.Close();

            using var fill = new SKPaint { 
                Color = (i % 2 == 0) ? SKColor.Parse("#2c3e50") : SKColor.Parse("#34495e"),
                IsAntialias = true 
            };
            canvas.DrawPath(path, fill);

            // Draw white divider lines
            using var borderPaint = new SKPaint { Color = SKColors.White, Style = SKPaintStyle.Stroke, StrokeWidth = 2, IsAntialias = true };
            canvas.DrawPath(path, borderPaint);

            // Draw Text (Rotated into slice) using Modern SKFont
            canvas.Save();
            canvas.RotateDegrees(i * sweepAngle + (sweepAngle / 2), center.X, center.Y);
            
            using var textPaint = new SKPaint { Color = SKColors.White, IsAntialias = true };
            using var font = new SKFont { Size = radius * 0.15f };
            
            // --- FIX FOR TITLE LOOKUP ---
            // Find the actual Movie object from the ViewModel using the MovieId
            var movieObj = ViewModel.Movies?.FirstOrDefault(m => m.Id == activeMovies[i].MovieId);
            string title = movieObj?.Title ?? "Unknown Movie";
            
            // Limit title length to prevent bleeding into other sections
            if(title.Length > 15) title = title.Substring(0, 15) + "...";

            canvas.DrawText(title, center.X + (radius * 0.5f), center.Y + (font.Size * 0.3f), SKTextAlign.Center, font, textPaint);
            canvas.Restore();
        }
        canvas.Restore();
    }

    private async Task SpinAsync()
    {
        var activeMovies = GetActiveMovies();
        if (_isSpinning || activeMovies.Count <= 1) return;

        _isSpinning = true;
        var random = new Random();
        _spinVelocity = 20f + (float)random.NextDouble() * 15f; // Initial kick

        while (_spinVelocity > 0.1f)
        {
            _rotationAngle = (_rotationAngle + _spinVelocity) % 360;
            _spinVelocity *= 0.97f; // Friction/Deceleration
            _canvasView?.Invalidate();
            await Task.Delay(16); // ~60 FPS
        }

        _isSpinning = false;
        
        // Re-render final resting place
        _canvasView?.Invalidate();
    }
}
